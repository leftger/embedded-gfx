7/// Configuration for screen-space water reflections (SSR).
8#[derive(Debug, Clone, Copy)]
9pub struct WaterReflectConfig<'a> {
10    /// Reference color buffer from the current or previous frame/field.
11    pub reference_buffer: &'a [Rgb565],
12    /// Buffer width in pixels.
13    pub width: usize,
14    /// Buffer height in pixels.
15    pub height: usize,
16    /// Screen row of the waterline/horizon axis.
17    pub waterline_y: i32,
18    /// Base water surface color.
19    pub surface_color: Rgb565,
20    /// Blend factor towards `surface_color` (0 = 100% reflection, 255 = 100% flat surface color).
21    pub alpha: u8,
22    /// Wave ripple displacement amplitude in pixels.
23    pub ripple_amplitude: i32,
24    /// Wave phase offset for animation.
25    pub ripple_phase: i32,
26}
27
28impl<'a> WaterReflectConfig<'a> {
29    /// Create a new SSR water reflection configuration.
30    pub fn new(
31        reference_buffer: &'a [Rgb565],
32        width: usize,
33        height: usize,
34        waterline_y: i32,
35        surface_color: Rgb565,
36        alpha: u8,
37    ) -> Self {
38        Self {
39            reference_buffer,
40            width,
41            height,
42            waterline_y,
43            surface_color,
44            alpha,
45            ripple_amplitude: 0,
46            ripple_phase: 0,
47        }
48    }
49
50    /// Set ripple wave parameters for animated water surface distortion.
51    pub fn with_ripples(mut self, amplitude: i32, phase: i32) -> Self {
52        self.ripple_amplitude = amplitude;
53        self.ripple_phase = phase;
54        self
55    }
56
57    /// Sample the mirrored pixel from the reference buffer with optional ripple distortion.
58    #[inline]
59    pub fn sample_reflection(&self, x: i32, y: i32) -> Rgb565 {
60        if self.reference_buffer.is_empty() || self.width == 0 || self.height == 0 {
61            return self.surface_color;
62        }
63
64        // Mirror formula: mirror_y = 2 * waterline_y - y
65        let raw_mirror_y = 2 * self.waterline_y - y;
66
67        // Apply simple integer ripple displacement if amplitude > 0
68        let (ripple_x, ripple_y) = if self.ripple_amplitude > 0 {
69            let offset_x = ((x + self.ripple_phase) & 7) - 3;
70            let offset_y = ((y + self.ripple_phase) & 7) - 3;
71            (
72                (offset_x * self.ripple_amplitude) / 4,
73                (offset_y * self.ripple_amplitude) / 4,
74            )
75        } else {
76            (0, 0)
77        };
78
79        let mirror_x = (x + ripple_x).clamp(0, self.width as i32 - 1) as usize;
80        let mirror_y = (raw_mirror_y + ripple_y).clamp(0, self.height as i32 - 1) as usize;
81
82        let idx = mirror_y * self.width + mirror_x;
83        if idx < self.reference_buffer.len() {
84            self.reference_buffer[idx]
85        } else {
86            self.surface_color
87        }
88    }
89}

91/// Fragment shader decorator for screen-space temporal water reflections.
92#[derive(Debug, Clone, Copy)]
93pub struct WaterReflectShader<'a, S> {
94    pub inner: S,
95    pub config: &'a WaterReflectConfig<'a>,
96}
97
98impl<'a, S: FragmentShader> FragmentShader for WaterReflectShader<'a, S> {
99    type Interpolants = S::Interpolants;
100
101    #[inline(always)]
102    fn shade(&self, x: i32, y: i32, z: ZDepth, interps: Self::Interpolants) -> Rgb565 {
103        let base_color = self.inner.shade(x, y, z, interps);
104        if y < self.config.waterline_y {
105            return base_color;
106        }
107
108        let reflected_color = self.config.sample_reflection(x, y);
109        if self.config.alpha == 0 {
110            reflected_color
111        } else if self.config.alpha == 255 {
112            base_color
113        } else {
114            fast_blend_rgb565(reflected_color, base_color, self.config.alpha)
115        }
116    }
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122    use crate::shader::FlatColorShader;
123    use embedded_graphics_core::pixelcolor::RgbColor;
124
125    #[test]
126    fn test_water_reflect_mirror_sampling() {
127        // 4x4 buffer where row 0 is RED, row 1 is GREEN, row 2 is BLUE, row 3 is WHITE
128        let mut buffer = [Rgb565::BLACK; 16];
129        for x in 0..4 {
130            buffer[0 * 4 + x] = Rgb565::RED;
131            buffer[1 * 4 + x] = Rgb565::GREEN;
132            buffer[2 * 4 + x] = Rgb565::BLUE;
133            buffer[3 * 4 + x] = Rgb565::WHITE;
134        }
135
136        // Waterline at row 2:
137        // For y = 2 -> mirror_y = 2*2 - 2 = 2 (BLUE)
138        // For y = 3 -> mirror_y = 2*2 - 3 = 1 (GREEN)
139        let config = WaterReflectConfig::new(&buffer, 4, 4, 2, Rgb565::CYAN, 0);
140        let shader = WaterReflectShader {
141            inner: FlatColorShader {
142                color: Rgb565::CYAN,
143            },
144            config: &config,
145        };
146
147        let color_above = shader.shade(0, 1, 0, ());
148        assert_eq!(color_above, Rgb565::CYAN); // Above waterline returns inner base color
149
150        let color_at_waterline = shader.shade(0, 2, 0, ());
151        assert_eq!(color_at_waterline, Rgb565::BLUE);
152
153        let color_below_waterline = shader.shade(0, 3, 0, ());
154        assert_eq!(color_below_waterline, Rgb565::GREEN);
155    }
