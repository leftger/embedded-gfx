1//! Atmospheric lens-flare effect with occlusion picking.
2//!
3//! Inspired by Jet's `LensFlare.hpp` and `Picking.hpp`. Projects a directional light source (sun)
4//! to screen space, checks occlusion via a single depth pick query, and renders a chain of
5//! optical flare elements along the sun-to-screen-center vector.
6

23/// A single flare element along the sun-to-screen-center axis.
24#[derive(Debug, Clone, Copy)]
25pub struct FlareElement {
26    /// Position along the sun -> center axis.
27    /// * `0.0`: pinned to the projected sun position.
28    /// * `0.5`: halfway between sun and screen center.
29    /// * `1.0`: at screen center.
30    /// * `-0.2`: 20% past the sun away from center.
31    /// * `1.3`: 30% past the center away from sun.
32    pub axis_t: f32,
33    /// Radius of the flare circle in screen pixels.
34    pub radius: i32,
35    /// Color of the flare element.
36    pub color: Rgb565,
37    /// Maximum opacity (0..=255).
38    pub base_alpha: u8,
39}
40
41impl FlareElement {
42    pub const fn new(axis_t: f32, radius: i32, color: Rgb565, base_alpha: u8) -> Self {
43        Self {
44            axis_t,
45            radius,
46            color,
47            base_alpha,
48        }
49    }
50}

52/// Atmospheric lens flare manager.
53pub struct LensFlare<const MAX_ELEMS: usize> {
54    /// World-space unit direction pointing toward the sun / light source.
55    pub sun_dir: Vector3<f32>,
56    /// Array of flare elements.
57    pub elements: Vec<FlareElement, MAX_ELEMS>,
58    /// Speed of fade in/out transitions (units per second).
59    pub fade_speed: f32,
60    /// Current smoothed visibility in 0.0..=1.0.
61    pub visibility: f32,
62    sun_screen: Option<(i32, i32)>,
63}
64
65impl<const MAX_ELEMS: usize> LensFlare<MAX_ELEMS> {
66    /// Create a new lens flare manager.
67    pub fn new(sun_dir: Vector3<f32>, fade_speed: f32) -> Self {
68        Self {
69            sun_dir: sun_dir.normalize(),
70            elements: Vec::new(),
71            fade_speed,
72            visibility: 0.0,
73            sun_screen: None,
74        }
75    }
76
77    /// Add a flare element to the chain.
78    pub fn add_element(&mut self, element: FlareElement) -> bool {
79        self.elements.push(element).is_ok()
80    }
81
82    /// Prepare the lens flare for the current frame.
83    ///
84    /// Projects the sun into screen coordinates. If visible on screen and in front of the camera,
85    /// returns a `PickQuery` to test for geometric occlusion.
86    pub fn prepare(
87        &mut self,
88        camera: &Camera,
89        screen_w: usize,
90        screen_h: usize,
91    ) -> Option<PickQuery> {
92        let sun_world = camera.position + self.sun_dir * 1000.0;
93        let vp = camera.vp_matrix;
94
95        let clip = vp * Point3::new(sun_world.x, sun_world.y, sun_world.z).to_homogeneous();
96        if clip.w <= 0.001 {
97            self.sun_screen = None;
98            return None;
99        }
100
101        let ndc_x = clip.x / clip.w;
102        let ndc_y = clip.y / clip.w;
103
104        // Check if projected sun is within the viewport bounds
105        if !(-1.2..=1.2).contains(&ndc_x) || !(-1.2..=1.2).contains(&ndc_y) {
106            self.sun_screen = None;
107            return None;
108        }
109
110        let sx = ((ndc_x * 0.5 + 0.5) * screen_w as f32) as i32;
111        let sy = ((1.0 - (ndc_y * 0.5 + 0.5)) * screen_h as f32) as i32;
112
113        self.sun_screen = Some((sx, sy));
114
115        if sx >= 0 && sx < screen_w as i32 && sy >= 0 && sy < screen_h as i32 {
116            Some(PickQuery::new(sx, sy))
117        } else {
118            None
119        }
120    }
121
122    /// Update the flare visibility state using the pick result from command execution.
123    pub fn update(&mut self, dt: f32, pick_hit: Option<&PickResult>) {
124        let target_visibility = if self.sun_screen.is_some() {
125            if pick_hit.is_none() {
126                1.0
127            } else {
128                0.0 // Occluded by scene geometry!
129            }
130        } else {
131            0.0
132        };
133
134        let delta = self.fade_speed * dt;
135        if self.visibility < target_visibility {
136            self.visibility = (self.visibility + delta).min(target_visibility);
137        } else if self.visibility > target_visibility {
138            self.visibility = (self.visibility - delta).max(target_visibility);
139        }
140    }
141
142    /// Render visible flare elements directly to the target framebuffer.
143    pub fn render<D>(&self, fb: &mut D, screen_w: usize, screen_h: usize) -> Result<(), D::Error>
144    where
145        D: DrawTarget<Color = Rgb565>,
146        D::Error: Debug,
147    {
148        if self.visibility <= 0.005 {
149            return Ok(());
150        }
151
152        let Some((sun_x, sun_y)) = self.sun_screen else {
153            return Ok(());
154        };
155
156        let center_x = screen_w as f32 * 0.5;
157        let center_y = screen_h as f32 * 0.5;
158        let dir_x = center_x - sun_x as f32;
159        let dir_y = center_y - sun_y as f32;
160
161        let w = screen_w as i32;
162        let h = screen_h as i32;
163
164        for elem in &self.elements {
165            let cx = (sun_x as f32 + dir_x * elem.axis_t) as i32;
166            let cy = (sun_y as f32 + dir_y * elem.axis_t) as i32;
167            let alpha = ((elem.base_alpha as f32) * self.visibility) as u8;
168            if alpha == 0 || elem.radius <= 0 {
169                continue;
170            }
171
172            let r = elem.radius;
173            let min_x = (cx - r).max(0);
174            let max_x = (cx + r).min(w - 1);
175            let min_y = (cy - r).max(0);
176            let max_y = (cy + r).min(h - 1);
177
178            let r_sq = r * r;
179            for py in min_y..=max_y {
180                let dy = py - cy;
181                for px in min_x..=max_x {
182                    let dx = px - cx;
183                    let dist_sq = dx * dx + dy * dy;
184                    if dist_sq <= r_sq {
185                        // Soft falloff towards edges
186                        let falloff = 1.0 - (dist_sq as f32 / r_sq as f32);
187                        let final_alpha = ((alpha as f32) * falloff) as u8;
188                        if final_alpha > 0 {
189                            let blended = fast_blend_rgb565(Rgb565::BLACK, elem.color, final_alpha);
190                            fb.draw_iter([Pixel(Point::new(px, py), blended)])?;
191                        }
192                    }
193                }
194            }
195        }
196
197        Ok(())
198    }
199}

206    struct MockFb {
207        pixels: [Rgb565; 400],
208    }
209
210    impl OriginDimensions for MockFb {
211        fn size(&self) -> Size {
212            Size::new(20, 20)
213        }
214    }
215
216    impl DrawTarget for MockFb {
217        type Color = Rgb565;
218        type Error = core::convert::Infallible;
219
220        fn draw_iter<I>(&mut self, pixels: I) -> Result<(), Self::Error>
221        where
222            I: IntoIterator<Item = Pixel<Self::Color>>,
223        {
224            for Pixel(point, color) in pixels {
225                if point.x >= 0 && point.x < 20 && point.y >= 0 && point.y < 20 {
226                    let idx = (point.y * 20 + point.x) as usize;
227                    self.pixels[idx] = color;
228                }
229            }
230            Ok(())
231        }
232    }

234    #[test]
235    fn test_lens_flare_lifecycle() {
236        let mut flare = LensFlare::<4>::new(Vector3::new(0.0, 0.0, -1.0), 10.0);
237        flare.add_element(FlareElement::new(0.0, 8, Rgb565::YELLOW, 255));
238        flare.add_element(FlareElement::new(0.5, 4, Rgb565::CYAN, 128));
239
240        let camera = Camera::new(320.0 / 240.0);
241
242        let query = flare.prepare(&camera, 320, 240);
243        assert!(query.is_some());
244        let q = query.unwrap();
245        // Camera looks towards -Z by default, so sun at (0,0,-1) maps to center
246        assert_eq!(q.x, 160);
247        assert_eq!(q.y, 120);
248
249        // Sun unoccluded -> visibility rises
250        flare.update(0.1, None);
251        assert!(flare.visibility > 0.0);
252
253        // Sun occluded -> visibility drops
254        let hit = PickResult {
255            x: 160,
256            y: 120,
257            depth: 100,
258            command_index: 0,
259        };
260        flare.update(0.2, Some(&hit));
261        assert_eq!(flare.visibility, 0.0);
262    }
