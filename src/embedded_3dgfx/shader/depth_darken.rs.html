1//! Cheap depth-based darkening ("Z brightness") for software renderers.
2//!
3//! This is a clean-room, no-allocation shading helper inspired by the general
4//! depth-darkening trick used by tiny embedded rasterisers. It darkens a
5//! fragment as it moves away from the camera, giving low-poly scenes extra
6//! depth without adding lights, fog buffers, or extra memory.
7
8use core::fmt::Debug;
9use embedded_graphics_core::pixelcolor::{Rgb565, RgbColor};
10
11/// Configuration for the depth-darkening effect.
12#[derive(Debug, Clone, Copy)]
13pub struct DepthDarkenConfig {
14    /// Maximum darkening amount in 0..=255. `0` disables the effect and `255`
15    /// fully darkens fragments at (or beyond) [`Self::far`].
16    pub max_darkness: u8,
17    /// Depth at which darkening begins (fixed-point 16.16).
18    pub near: u32,
19    /// Depth at which maximum darkening is reached (fixed-point 16.16).
20    pub far: u32,
21}
22
23impl DepthDarkenConfig {
24    /// Creates a new config from scene-space distances.
25    ///
26    /// Distances use the same units as the scene depth buffer and are converted
27    /// internally to 16.16 fixed point.
28    pub fn new(max_darkness: u8, near: f32, far: f32) -> Self {
29        Self {
30            max_darkness,
31            near: (near * 65536.0) as u32,
32            far: (far * 65536.0) as u32,
33        }
34    }
35
36    /// Applies depth darkening to `base_color` for a given depth value.
37    #[inline]
38    pub fn apply(&self, base_color: Rgb565, depth: u32) -> Rgb565 {
39        if self.max_darkness == 0 {
40            return base_color;
41        }
42
43        // 0 at near, 65536 at far, expressed in 16.16 fixed point.
44        let t = if depth <= self.near {
45            0u32
46        } else if depth >= self.far {
47            65536u32
48        } else {
49            let numerator = (depth - self.near) as u64;
50            let denominator = (self.far - self.near).max(1) as u64;
51            ((numerator * 65536) / denominator) as u32
52        };
53
54        // Scale = 1.0 - (t * max_darkness / 255), again in 16.16 fixed point.
55        let dark = ((t as u64 * self.max_darkness as u64) / 255) as u32;
56        let scale = 65536u32.saturating_sub(dark);
57
58        let r = ((base_color.r() as u32 * scale) / 65536) as u8;
59        let g = ((base_color.g() as u32 * scale) / 65536) as u8;
60        let b = ((base_color.b() as u32 * scale) / 65536) as u8;
61
62        Rgb565::new(r, g, b)
63    }
64}

66/// Fragment shader decorator that darkens fragments by depth.
67#[derive(Debug, Clone, Copy)]
68pub struct DepthDarkenShader<'a, S> {
69    /// Inner shader whose output is darkend.
70    pub inner: S,
71    /// Darkening configuration.
72    pub config: &'a DepthDarkenConfig,
73}
74
75impl<'a, S: super::FragmentShader> super::FragmentShader for DepthDarkenShader<'a, S> {
76    type Interpolants = S::Interpolants;
77
78    #[inline(always)]
79    fn shade(&self, x: i32, y: i32, z: crate::ZDepth, interps: Self::Interpolants) -> Rgb565 {
80        let base = self.inner.shade(x, y, z, interps);
81        self.config.apply(base, u32::from(z))
82    }
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88    #[cfg(not(feature = "depth-u16"))]
89    use crate::shader::{FlatColorShader, FragmentShader};
90
91    #[test]
92    fn test_depth_darken_interpolation() {
93        let config = DepthDarkenConfig::new(255, 1.0, 10.0);
94        let color = Rgb565::WHITE;
95
96        // Near or closer stays unchanged.
97        assert_eq!(config.apply(color, (1.0 * 65536.0) as u32), color);
98        assert_eq!(config.apply(color, 0), color);
99
100        // Far or beyond becomes black at max darkness.
101        assert_eq!(config.apply(color, (10.0 * 65536.0) as u32), Rgb565::BLACK);
102        assert_eq!(config.apply(color, (20.0 * 65536.0) as u32), Rgb565::BLACK);
103
104        // Halfway at 50% max darkness should be visibly between the endpoints.
105        let mid = config.apply(color, (5.5 * 65536.0) as u32);
106        assert!(mid.r() > 0 && mid.r() < 20);
107        assert!(mid.g() > 0 && mid.g() < 40);
108    }
109
110    #[test]
111    fn test_zero_max_darkness_is_identity() {
112        let config = DepthDarkenConfig::new(0, 1.0, 10.0);
113        let color = Rgb565::new(10, 20, 30);
114        assert_eq!(config.apply(color, (100.0 * 65536.0) as u32), color);
115    }
116
117    #[test]
118    #[cfg(not(feature = "depth-u16"))]
119    fn test_depth_darken_shader_decorator() {
120        let config = DepthDarkenConfig::new(255, 1.0, 5.0);
121        let inner = FlatColorShader { color: Rgb565::RED };
122        let shader = DepthDarkenShader {
123            inner,
124            config: &config,
125        };
126
127        let near = crate::to_zdepth(65536);
128        let far = crate::to_zdepth(5 * 65536);
129        assert_eq!(shader.shade(0, 0, near, ()), Rgb565::RED);
130        assert_eq!(shader.shade(0, 0, far, ()), Rgb565::BLACK);
131    }
