4/// Configuration for depth-based fog effect
5#[derive(Debug, Clone, Copy)]
6pub struct FogConfig {
7    /// Fog color to blend towards
8    pub color: Rgb565,
9    /// Near plane distance (fixed-point 16.16 format)
10    pub near: u32,
11    /// Far plane distance (fixed-point 16.16 format)
12    pub far: u32,
13}
14
15impl FogConfig {
16    /// Create a new fog configuration
17    ///
18    /// # Arguments
19    /// * `color` - The fog color
20    /// * `near` - Near distance (depth values closer than this have no fog)
21    /// * `far` - Far distance (depth values farther than this are fully fogged)
22    pub fn new(color: Rgb565, near: f32, far: f32) -> Self {
23        Self {
24            color,
25            near: (near * 65536.0) as u32,
26            far: (far * 65536.0) as u32,
27        }
28    }
29
30    /// Apply fog effect to a color based on depth
31    #[inline]
32    pub fn apply(&self, base_color: Rgb565, depth: u32) -> Rgb565 {
33        // Calculate fog factor: 0.0 at near plane, 1.0 at far plane
34        let fog_factor = if depth <= self.near {
35            0u32
36        } else if depth >= self.far {
37            65536u32 // 1.0 in fixed-point
38        } else {
39            // Linear interpolation: (depth - near) / (far - near)
40            let numerator = (depth - self.near) as u64;
41            let denominator = (self.far - self.near) as u64;
42            ((numerator * 65536) / denominator) as u32
43        };
44
45        // Blend base color with fog color
46        let base_r = base_color.r() as u32;
47        let base_g = base_color.g() as u32;
48        let base_b = base_color.b() as u32;
49
50        let fog_r = self.color.r() as u32;
51        let fog_g = self.color.g() as u32;
52        let fog_b = self.color.b() as u32;
53
54        // fog_factor is in 16.16 fixed-point format
55        let r = ((base_r * (65536 - fog_factor) + fog_r * fog_factor) / 65536) as u8;
56        let g = ((base_g * (65536 - fog_factor) + fog_g * fog_factor) / 65536) as u8;
57        let b = ((base_b * (65536 - fog_factor) + fog_b * fog_factor) / 65536) as u8;
58
59        Rgb565::new(r, g, b)
60    }
61}

63/// Fragment shader decorator that applies depth-based fog.
64#[derive(Debug, Clone, Copy)]
65pub struct FogShader<'a, S> {
66    pub inner: S,
67    pub fog: &'a FogConfig,
68}

70impl<'a, S: super::FragmentShader> super::FragmentShader for FogShader<'a, S> {
71    type Interpolants = S::Interpolants;
72
73    #[inline(always)]
74    fn shade(&self, x: i32, y: i32, z: crate::ZDepth, interps: Self::Interpolants) -> Rgb565 {
75        let base = self.inner.shade(x, y, z, interps);
