13/// Hue-Saturation-Value (HSV) color representation.
14///
15/// * `h`: Hue in degrees `[0.0, 360.0)`.
16/// * `s`: Saturation in `[0.0, 1.0]`.
17/// * `v`: Value / Brightness in `[0.0, 1.0]`.
18#[derive(Debug, Clone, Copy, PartialEq, Default)]
19pub struct Hsv {
20    pub h: f32,
21    pub s: f32,
22    pub v: f32,
23}
24
25impl Hsv {
26    /// Create a new HSV color.
27    #[inline]
28    pub const fn new(h: f32, s: f32, v: f32) -> Self {
29        Self { h, s, v }
30    }
31
32    /// Convert [`Rgb565`] to [`Hsv`].
33    pub fn from_rgb565(rgb: Rgb565) -> Self {
34        let r = rgb.r() as f32 / 31.0;
35        let g = rgb.g() as f32 / 63.0;
36        let b = rgb.b() as f32 / 31.0;
37
38        let max = r.max(g).max(b);
39        let min = r.min(g).min(b);
40        let delta = max - min;
41
42        let v = max;
43        let s = if max > 1e-5 { delta / max } else { 0.0 };
44
45        let h = if delta < 1e-5 {
46            0.0
47        } else if (max - r).abs() < 1e-5 {
48            60.0 * (((g - b) / delta) % 6.0)
49        } else if (max - g).abs() < 1e-5 {
50            60.0 * (((b - r) / delta) + 2.0)
51        } else {
52            60.0 * (((r - g) / delta) + 4.0)
53        };
54
55        let h = if h < 0.0 { h + 360.0 } else { h };
56
57        Self { h, s, v }
58    }
59
60    /// Convert [`Hsv`] to [`Rgb565`].
61    pub fn to_rgb565(&self) -> Rgb565 {
62        let c = self.v * self.s;
63        let h_prime = (self.h.rem_euclid(360.0)) / 60.0;
64        let x = c * (1.0 - ((h_prime % 2.0) - 1.0).abs());
65        let m = self.v - c;
66
67        let (r1, g1, b1) = if h_prime < 1.0 {
68            (c, x, 0.0)
69        } else if h_prime < 2.0 {
70            (x, c, 0.0)
71        } else if h_prime < 3.0 {
72            (0.0, c, x)
73        } else if h_prime < 4.0 {
74            (0.0, x, c)
75        } else if h_prime < 5.0 {
76            (x, 0.0, c)
77        } else {
78            (c, 0.0, x)
79        };
80
81        let r = (((r1 + m) * 31.0).round() as u8).min(31);
82        let g = (((g1 + m) * 63.0).round() as u8).min(63);
83        let b = (((b1 + m) * 31.0).round() as u8).min(31);
84
85        Rgb565::new(r, g, b)
86    }
87
88    /// Interpolate between two HSV colors along the shortest arc of the hue circle.
89    ///
90    /// Avoids the muddy grayish desaturation that occurs when interpolating directly in RGB.
91    pub fn lerp(a: Self, b: Self, t: f32) -> Self {
92        let t = t.clamp(0.0, 1.0);
93
94        // Shortest hue delta on 360-degree circle
95        let mut d_h = b.h - a.h;
96        if d_h > 180.0 {
97            d_h -= 360.0;
98        } else if d_h < -180.0 {
99            d_h += 360.0;
100        }
101
102        let h = (a.h + d_h * t).rem_euclid(360.0);
103        let s = a.s + (b.s - a.s) * t;
104        let v = a.v + (b.v - a.v) * t;
105
106        Self { h, s, v }
107    }
108}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn test_hsv_roundtrip_primaries() {
136        let red = Rgb565::new(31, 0, 0);
137        let hsv_red = Hsv::from_rgb565(red);
138        assert!((hsv_red.h - 0.0).abs() < 1e-2 || (hsv_red.h - 360.0).abs() < 1e-2);
139        assert!((hsv_red.s - 1.0).abs() < 1e-2);
140        assert!((hsv_red.v - 1.0).abs() < 1e-2);
141        assert_eq!(hsv_red.to_rgb565(), red);
142
143        let green = Rgb565::new(0, 63, 0);
144        let hsv_green = Hsv::from_rgb565(green);
145        assert!((hsv_green.h - 120.0).abs() < 1e-2);
146        assert_eq!(hsv_green.to_rgb565(), green);
147
148        let blue = Rgb565::new(0, 0, 31);
149        let hsv_blue = Hsv::from_rgb565(blue);
150        assert!((hsv_blue.h - 240.0).abs() < 1e-2);
151        assert_eq!(hsv_blue.to_rgb565(), blue);
152    }
153
154    #[test]
155    fn test_hsv_lerp_shortest_arc() {
156        // 350 deg (near red) to 10 deg (near red) should go through 0 deg, not 180 deg
157        let a = Hsv::new(350.0, 1.0, 1.0);
158        let b = Hsv::new(10.0, 1.0, 1.0);
159        let mid = Hsv::lerp(a, b, 0.5);
160        assert!((mid.h - 0.0).abs() < 1e-2 || (mid.h - 360.0).abs() < 1e-2);
161    }
