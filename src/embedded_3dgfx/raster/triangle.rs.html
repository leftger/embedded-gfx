2use embedded_graphics_core::draw_target::DrawTarget;
3use embedded_graphics_core::pixelcolor::Rgb565;
4use embedded_graphics_core::prelude::Point;

10/// Screen-space 2D triangle area (cross product).
11#[inline(always)]
12pub fn tri_area2(a: Point, b: Point, c: Point) -> i32 {
13    (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
14}
15
16/// Generic depth-buffered triangle rasterizer with composable fragment shader.
17///
18/// Sweeps scanlines top-to-bottom, interpolates depth (16.16 fixed-point or float),
19/// and delegates pixel shading to the zero-cost generic `shader`.
20pub fn draw_triangle_zbuffered<D, S>(
21    points: [Point; 3],
22    depths: [f32; 3],
23    shader: &S,
24    fb: &mut D,
25    zbuffer: &mut [ZDepth],
26) where
27    D: DrawTarget<Color = Rgb565>,
28    <D as DrawTarget>::Error: Debug,
29    S: FragmentShader<Interpolants = ()>,
30{
31    // Sort vertices by Y coordinate (p1.y <= p2.y <= p3.y)
32    let mut indices = [0, 1, 2];
33    if points[indices[0]].y > points[indices[1]].y {
34        indices.swap(0, 1);
35    }
36    if points[indices[1]].y > points[indices[2]].y {
37        indices.swap(1, 2);
38    }
39    if points[indices[0]].y > points[indices[1]].y {
40        indices.swap(0, 1);
41    }
42
43    let p1 = points[indices[0]];
44    let p2 = points[indices[1]];
45    let p3 = points[indices[2]];
46
47    let z1 = depths[indices[0]];
48    let z2 = depths[indices[1]];
49    let z3 = depths[indices[2]];
50
51    let total_height = p3.y - p1.y;
52    if total_height == 0 {
53        return;
54    }
55
56    let area = tri_area2(p1, p2, p3);
57    if area == 0 {
58        return;
59    }
60
61    let target_width = fb.bounding_box().size.width as i32;
62    let target_height = fb.bounding_box().size.height as i32;
63
64    for y in p1.y..=p3.y {
65        if y < 0 || y >= target_height {
66            continue;
67        }
68
69        let second_half = y > p2.y || p2.y == p1.y;
70        let segment_height = if second_half {
71            p3.y - p2.y
72        } else {
73            p2.y - p1.y
74        };
75        if segment_height == 0 {
76            continue;
77        }
78
79        let edge1 = EdgeStepper::new(p1, p3, y);
80        let edge2 = if second_half {
81            EdgeStepper::new(p2, p3, y)
82        } else {
83            EdgeStepper::new(p1, p2, y)
84        };
85
86        let mut x1 = edge1.current_x();
87        let mut x2 = edge2.current_x();
88
89        if x1 > x2 {
90            core::mem::swap(&mut x1, &mut x2);
91        }
92
93        let min_x = x1.max(0);
94        let max_x = x2.min(target_width - 1);
95
96        for x in min_x..=max_x {
97            // Barycentric depth interpolation
98            let w1 = tri_area2(p2, p3, Point::new(x, y));
99            let w2 = tri_area2(p3, p1, Point::new(x, y));
100            let w3 = tri_area2(p1, p2, Point::new(x, y));
101
102            if area > 0 && (w1 < 0 || w2 < 0 || w3 < 0) {
103                continue;
104            }
105            if area < 0 && (w1 > 0 || w2 > 0 || w3 > 0) {
106                continue;
107            }
108
109            let inv_area = 1.0 / (area as f32);
110            let alpha = (w1 as f32) * inv_area;
111            let beta = (w2 as f32) * inv_area;
112            let gamma = (w3 as f32) * inv_area;
113
114            let interp_z = alpha * z1 + beta * z2 + gamma * z3;
115            if interp_z < 0.0 {
116                continue;
117            }
118
119            let z_int = crate::to_zdepth((interp_z * 65536.0) as u32);
120            let zb_idx = (y as usize) * (target_width as usize) + (x as usize);
121
122            if zb_idx < zbuffer.len() && z_int < zbuffer[zb_idx] {
123                zbuffer[zb_idx] = z_int;
124                let color = shader.shade(x, y, z_int, ());
125                fb.draw_iter([embedded_graphics_core::Pixel(Point::new(x, y), color)])
126                    .unwrap();
127            }
128        }
129    }
130}
