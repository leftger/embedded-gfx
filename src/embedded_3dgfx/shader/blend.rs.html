3/// Fast RGB565 alpha blending.
4///
5/// Blends `fg` color over `bg` color using an 8-bit alpha value `alpha` ∈ [0, 255].
6#[inline(always)]
7pub fn fast_blend_rgb565(bg: Rgb565, fg: Rgb565, alpha: u8) -> Rgb565 {
8    if alpha == 255 {
9        return fg;
10    }
11    if alpha == 0 {
12        return bg;
13    }
14    let a = alpha as u32;
15    let inv = 255 - a;
16    let r = (bg.r() as u32 * inv + fg.r() as u32 * a) / 255;
17    let g = (bg.g() as u32 * inv + fg.g() as u32 * a) / 255;
18    let b = (bg.b() as u32 * inv + fg.b() as u32 * a) / 255;
19    Rgb565::new(r as u8, g as u8, b as u8)
20}

22/// Fast RGBA8888 alpha blending.
23///
24/// Blends `fg` RGBA8888 channel tuple over `bg` RGBA8888 channel tuple.
25#[inline(always)]
26pub fn fast_blend_rgba8888(bg: [u8; 4], fg: [u8; 4]) -> [u8; 4] {
27    let a = fg[3] as u32;
28    if a == 255 {
29        return fg;
30    }
31    if a == 0 {
32        return bg;
33    }
34    let inv = 255 - a;
35    let r = (bg[0] as u32 * inv + fg[0] as u32 * a) / 255;
36    let g = (bg[1] as u32 * inv + fg[1] as u32 * a) / 255;
37    let b = (bg[2] as u32 * inv + fg[2] as u32 * a) / 255;
38    let out_a = fg[3] as u32 + (bg[3] as u32 * inv) / 255;
39    [r as u8, g as u8, b as u8, out_a as u8]
40}

42/// Fast RGBA8888 to RGB565 alpha blending.
43///
44/// Blends an RGBA8888 foreground pixel directly onto an RGB565 background pixel.
45#[inline(always)]
46pub fn fast_blend_rgba8888_to_rgb565(bg: Rgb565, fg_rgba: [u8; 4]) -> Rgb565 {
47    let alpha = fg_rgba[3];
48    if alpha == 0 {
49        return bg;
50    }
51    let fg_r = fg_rgba[0] >> 3;
52    let fg_g = fg_rgba[1] >> 2;
53    let fg_b = fg_rgba[2] >> 3;
54
55    if alpha == 255 {
56        return Rgb565::new(fg_r, fg_g, fg_b);
57    }
58
59    let fg_color = Rgb565::new(fg_r, fg_g, fg_b);
60    fast_blend_rgb565(bg, fg_color, alpha)
61}

63/// Fast color inversion (reverse color) filter for RGB565.
64///
65/// Inverts RGB color channels.
66#[inline(always)]
67pub fn reverse_color_rgb565(c: Rgb565) -> Rgb565 {
68    Rgb565::new(31 - c.r(), 63 - c.g(), 31 - c.b())
69}
