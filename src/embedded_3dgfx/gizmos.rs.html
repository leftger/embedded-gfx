30/// Emit 12 wireframe edges for a model-space AABB transformed by `model_matrix`.
31///
32/// Vertices are projected with `vp * model`. Degenerate (behind-camera) edges
33/// are skipped.
34pub fn emit_aabb_wireframe<F>(
35    aabb: &Aabb,
36    model_matrix: &Matrix4<f32>,
37    vp_matrix: &Matrix4<f32>,
38    color: Rgb565,
39    mut emit: F,
40) where
41    F: FnMut(DrawPrimitive),
42{
43    let mvp = vp_matrix * model_matrix;
44    let corners = aabb.corners();
45    let mut screen: [Option<Point2<i32>>; 8] = [None; 8];
46    for (i, c) in corners.iter().enumerate() {
47        let clip = mvp * nalgebra::Vector4::new(c.x, c.y, c.z, 1.0);
48        if clip.w <= 0.0 {
49            continue;
50        }
51        let inv_w = 1.0 / clip.w;
52        // NDC → placeholder pixel space is caller-agnostic; use raw NDC*1000
53        // style ints only if engine projection already baked screen scale.
54        // Prefer transforming like the engine: assume vp already maps to pixels
55        // when used with K3dengine (it does — see transform_point).
56        screen[i] = Some(Point2::new(
57            (clip.x * inv_w) as i32,
58            (clip.y * inv_w) as i32,
59        ));
60    }
61    for &(a, b) in &AABB_EDGES {
62        if let (Some(pa), Some(pb)) = (screen[a], screen[b]) {
63            emit(DrawPrimitive::Line([pa, pb], color));
64        }
65    }
66}

68/// Emit AABB wireframe using the engine's screen-space projection path.
69pub fn emit_aabb_wireframe_projected<F>(
70    aabb: &Aabb,
71    model_matrix: &Matrix4<f32>,
72    project: impl Fn([f32; 3]) -> Option<Point3<i32>>,
73    color: Rgb565,
74    mut emit: F,
75) where
76    F: FnMut(DrawPrimitive),
77{
78    let corners = aabb.corners();
79    let mut screen: [Option<Point2<i32>>; 8] = [None; 8];
80    for (i, c) in corners.iter().enumerate() {
81        let world = model_matrix.transform_point(&Point3::new(c.x, c.y, c.z));
82        screen[i] = project([world.x, world.y, world.z]).map(|p| p.xy());
83    }
84    for &(a, b) in &AABB_EDGES {
85        if let (Some(pa), Some(pb)) = (screen[a], screen[b]) {
86            emit(DrawPrimitive::Line([pa, pb], color));
87        }
88    }
89}

91/// Approximate frustum corner rays at `near` and `far` along the view axes,
92/// then emit the 12 edges of the view frustum wireframe.
93///
94/// Uses the camera's FOV/aspect/near/far rather than extracting planes from
95/// `vp_matrix`, which keeps the math small and deterministic on MCU.
96pub fn emit_frustum_wireframe<F>(
97    camera: &Camera,
98    project: impl Fn([f32; 3]) -> Option<Point3<i32>>,
99    color: Rgb565,
100    mut emit: F,
101) where
102    F: FnMut(DrawPrimitive),
103{
104    let aspect = camera.get_aspect_ratio();
105    // Reconstruct eye-space frustum corners from near/far and a synthetic FOV.
106    // Camera stores fov privately; approximate from projection by sampling.
107    // Use a fixed vertical FOV estimate via near-plane height from vp if needed.
108    // Fallback: 90° vertical FOV when we cannot read fov — Camera exposes no getter.
109    // We use look direction + orthonormal basis from the view matrix.
110    let view = camera.view_matrix;
111    // view maps world→eye; inverse rotation is view^T for orthonormal view.
112    let r = view.fixed_view::<3, 3>(0, 0);
113    let right = Vector3::new(r[(0, 0)], r[(0, 1)], r[(0, 2)]);
114    let up = Vector3::new(r[(1, 0)], r[(1, 1)], r[(1, 2)]);
115    let forward = -Vector3::new(r[(2, 0)], r[(2, 1)], r[(2, 2)]); // camera looks down -Z in eye space
116
117    // Derive half-angles from the camera's vertical FOV.
118    let half_v = (camera.fovy() * 0.5).tan();
119    let half_h = half_v * aspect;
120
121    let eye = camera.position.coords;
122    let mut corners = [Vector3::zeros(); 8];
123    for (i, &z) in [camera.near, camera.far].iter().enumerate() {
124        let center = eye + forward * z;
125        let h = half_h * z;
126        let v = half_v * z;
127        let base = i * 4;
128        corners[base] = center - right * h - up * v;
129        corners[base + 1] = center + right * h - up * v;
130        corners[base + 2] = center + right * h + up * v;
131        corners[base + 3] = center - right * h + up * v;
132    }
133
134    let mut screen: [Option<Point2<i32>>; 8] = [None; 8];
135    for (i, c) in corners.iter().enumerate() {
136        screen[i] = project([c.x, c.y, c.z]).map(|p| p.xy());
137    }
138    for &(a, b) in &AABB_EDGES {
139        if let (Some(pa), Some(pb)) = (screen[a], screen[b]) {
140            emit(DrawPrimitive::Line([pa, pb], color));
141        }
142    }
143}

145/// Emit a wireframe grid on the XZ plane centered at `center`.
146///
147/// * `center`: World-space center point of the grid.
148/// * `cell_size`: Distance between adjacent grid lines.
149/// * `half_count`: Number of grid cells in positive and negative directions (total lines: `2 * half_count + 1` per axis).
150/// * `grid_color`: Line color for standard grid lines.
151/// * `axis_color`: Optional highlight color for the primary center axes passing through `center`.
152/// * `project`: World-to-screen projection closure.
153/// * `emit`: Callback receiving emitted line primitives.
154pub fn emit_ground_grid<F>(
155    center: Point3<f32>,
156    cell_size: f32,
157    half_count: i32,
158    grid_color: Rgb565,
159    axis_color: Option<Rgb565>,
160    project: impl Fn([f32; 3]) -> Option<Point3<i32>>,
161    mut emit: F,
162) where
163    F: FnMut(DrawPrimitive),
164{
165    let extent = half_count as f32 * cell_size;
166    let min_x = center.x - extent;
167    let max_x = center.x + extent;
168    let min_z = center.z - extent;
169    let max_z = center.z + extent;
170    let y = center.y;
171
172    // Lines parallel to Z (varying X)
173    for i in -half_count..=half_count {
174        let x = center.x + i as f32 * cell_size;
175        let color = if i == 0 {
176            axis_color.unwrap_or(grid_color)
177        } else {
178            grid_color
179        };
180
181        if let (Some(p0), Some(p1)) = (project([x, y, min_z]), project([x, y, max_z])) {
182            emit(DrawPrimitive::Line([p0.xy(), p1.xy()], color));
183        }
184    }
185
186    // Lines parallel to X (varying Z)
187    for j in -half_count..=half_count {
188        let z = center.z + j as f32 * cell_size;
189        let color = if j == 0 {
190            axis_color.unwrap_or(grid_color)
191        } else {
192            grid_color
193        };
194
195        if let (Some(p0), Some(p1)) = (project([min_x, y, z]), project([max_x, y, z])) {
196            emit(DrawPrimitive::Line([p0.xy(), p1.xy()], color));
197        }
198    }
199}

201/// Emit 3D RGB coordinate axes (Red = +X, Green = +Y, Blue = +Z) at `origin`.
202///
203/// * `origin`: World-space pivot point.
204/// * `model_matrix`: Model-to-world transform matrix.
205/// * `length`: Length of each axis line.
206/// * `project`: World-to-screen projection closure.
207/// * `emit`: Callback receiving emitted line primitives.
208pub fn emit_transform_axes<F>(
209    origin: Point3<f32>,
210    model_matrix: &Matrix4<f32>,
211    length: f32,
212    project: impl Fn([f32; 3]) -> Option<Point3<i32>>,
213    mut emit: F,
214) where
215    F: FnMut(DrawPrimitive),
216{
217    let red = Rgb565::new(31, 0, 0);
218    let green = Rgb565::new(0, 63, 0);
219    let blue = Rgb565::new(0, 0, 31);
220
221    let p_orig = model_matrix.transform_point(&origin);
222    let p_x = model_matrix.transform_point(&(origin + Vector3::new(length, 0.0, 0.0)));
223    let p_y = model_matrix.transform_point(&(origin + Vector3::new(0.0, length, 0.0)));
224    let p_z = model_matrix.transform_point(&(origin + Vector3::new(0.0, 0.0, length)));
225
226    if let (Some(s0), Some(sx)) = (
227        project([p_orig.x, p_orig.y, p_orig.z]),
228        project([p_x.x, p_x.y, p_x.z]),
229    ) {
230        emit(DrawPrimitive::Line([s0.xy(), sx.xy()], red));
231    }
232    if let (Some(s0), Some(sy)) = (
233        project([p_orig.x, p_orig.y, p_orig.z]),
234        project([p_y.x, p_y.y, p_y.z]),
235    ) {
236        emit(DrawPrimitive::Line([s0.xy(), sy.xy()], green));
237    }
238    if let (Some(s0), Some(sz)) = (
239        project([p_orig.x, p_orig.y, p_orig.z]),
240        project([p_z.x, p_z.y, p_z.z]),
241    ) {
242        emit(DrawPrimitive::Line([s0.xy(), sz.xy()], blue));
243    }
244}

246/// Emit 2D screen-space text lines using the Hershey simplex stroke font.
247///
248/// * `text`: ASCII string (characters 32..=126).
249/// * `origin`: Screen-space top-left coordinate.
250/// * `scale`: Scaling factor applied to glyphs (height ≈ `21.0 * scale` pixels).
251/// * `color`: Line color.
252/// * `emit`: Callback receiving emitted line primitives.
253pub fn emit_stroke_text_2d<F>(
254    text: &str,
255    origin: Point2<i32>,
256    scale: f32,
257    color: Rgb565,
258    mut emit: F,
259) where
260    F: FnMut(DrawPrimitive),
261{
262    use crate::simplex_stroke_font::{SIMPLEX_CAP_HEIGHT, SIMPLEX_STROKE_FONT};
263
264    let mut cursor_x = origin.x as f32;
265    let baseline_y = origin.y as f32 + SIMPLEX_CAP_HEIGHT * scale;
266
267    for c in text.chars() {
268        if c == ' ' {
269            cursor_x += SIMPLEX_STROKE_FONT.advance as f32 * scale;
270            continue;
271        }
272
273        if let Some((advance, stroke_range)) = SIMPLEX_STROKE_FONT.get_glyph(c) {
274            for s in stroke_range {
275                let point_range = SIMPLEX_STROKE_FONT.strokes[s].clone();
276                if point_range.len() < 2 {
277                    continue;
278                }
279
280                let mut prev: Option<Point2<i32>> = None;
281                for p_idx in point_range {
282                    let [px, py] = SIMPLEX_STROKE_FONT.positions[p_idx];
283                    let sx = (cursor_x + px as f32 * scale) as i32;
284                    let sy = (baseline_y - py as f32 * scale) as i32;
285                    let curr = Point2::new(sx, sy);
286
287                    if let Some(p) = prev {
288                        emit(DrawPrimitive::Line([p, curr], color));
289                    }
290                    prev = Some(curr);
291                }
292            }
293            cursor_x += advance as f32 * scale;
294        } else {
295            cursor_x += SIMPLEX_STROKE_FONT.advance as f32 * scale;
296        }
297    }
298}

300/// Emit 3D text in world space projected to screen lines using the Hershey simplex stroke font.
301///
302/// Glyphs are placed on the XY plane in world space (advancing along +X, with +Y up)
303/// starting at `origin`, and projected to screen space.
304///
305/// * `text`: ASCII string (characters 32..=126).
306/// * `origin`: World-space position of first glyph baseline.
307/// * `scale`: World scale factor for glyphs (height in world units ≈ `21.0 * scale`).
308/// * `color`: Line color.
309/// * `project`: World-to-screen projection closure.
310/// * `emit`: Callback receiving emitted line primitives.
311pub fn emit_stroke_text_projected<F>(
312    text: &str,
313    origin: Point3<f32>,
314    scale: f32,
315    color: Rgb565,
316    project: impl Fn([f32; 3]) -> Option<Point3<i32>>,
317    mut emit: F,
318) where
319    F: FnMut(DrawPrimitive),
320{
321    use crate::simplex_stroke_font::SIMPLEX_STROKE_FONT;
322
323    let mut cursor_x = origin.x;
324
325    for c in text.chars() {
326        if c == ' ' {
327            cursor_x += SIMPLEX_STROKE_FONT.advance as f32 * scale;
328            continue;
329        }
330
331        if let Some((advance, stroke_range)) = SIMPLEX_STROKE_FONT.get_glyph(c) {
332            for s in stroke_range {
333                let point_range = SIMPLEX_STROKE_FONT.strokes[s].clone();
334                if point_range.len() < 2 {
335                    continue;
336                }
337
338                let mut prev: Option<Point2<i32>> = None;
339                for p_idx in point_range {
340                    let [px, py] = SIMPLEX_STROKE_FONT.positions[p_idx];
341                    let wx = cursor_x + px as f32 * scale;
342                    let wy = origin.y + py as f32 * scale;
343                    let wz = origin.z;
344
345                    let curr = project([wx, wy, wz]).map(|p| p.xy());
346
347                    if let (Some(p), Some(c)) = (prev, curr) {
348                        emit(DrawPrimitive::Line([p, c], color));
349                    }
350                    prev = curr;
351                }
352            }
353            cursor_x += advance as f32 * scale;
354        } else {
355            cursor_x += SIMPLEX_STROKE_FONT.advance as f32 * scale;
356        }
357    }
358}

365    #[test]
366    fn aabb_emits_up_to_twelve_lines() {
367        let aabb = Aabb::from_min_max(Vector3::new(-1.0, -1.0, -1.0), Vector3::new(1.0, 1.0, 1.0));
368        let mut count = 0;
369        emit_aabb_wireframe_projected(
370            &aabb,
371            &Matrix4::identity(),
372            |p| Some(Point3::new(p[0] as i32, p[1] as i32, 1)),
373            Rgb565::CSS_LIME,
374            |_| count += 1,
375        );
376        assert_eq!(count, 12);
377    }
378
379    #[test]
380    fn test_emit_ground_grid() {
381        let mut line_count = 0;
382        emit_ground_grid(
383            Point3::new(0.0, 0.0, 0.0),
384            1.0,
385            2,
386            Rgb565::CSS_GRAY,
387            Some(Rgb565::CSS_WHITE),
388            |p| Some(Point3::new(p[0] as i32, p[2] as i32, 1)),
389            |_| line_count += 1,
390        );
391        // 2 * half_count + 1 = 5 lines per axis * 2 axes = 10 lines
392        assert_eq!(line_count, 10);
393    }
